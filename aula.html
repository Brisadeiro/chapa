const express = require('express') const app = express() app.use(express.urlencoded({ extended: true })) app.get('/', (request, response) => { response.send('Envie os parâmetros n1 e n2 para as rotas: /soma, /subtracao, /multiplicacao ou /divisao') }) app.get('/soma', (req, res) => { console.log("n1=" + req.query.n1) console.log("n2=" + req.query.n2) let n1 = parseFloat(req.query.n1) let n2 = parseFloat(req.query.n2) let result = n1 + n2 if (isNaN(result)) { result = "Valores inválidos." } else { result = `${req.query.n1} + ${req.query.n2} = ${result}` } res.send({ conta: result }) }) app.get('/subtracao', (req, res) => { console.log("n1=" + req.query.n1) console.log("n2=" + req.query.n2) let n1 = parseFloat(req.query.n1) let n2 = parseFloat(req.query.n2) let result = n1 - n2 if (isNaN(result)) { result = "Valores inválidos." } else { result = `${req.query.n1} - ${req.query.n2} = ${result}` } res.send({ conta: result }) }) app.get('/multiplicacao', (req, res) => { console.log("n1=" + req.query.n1) console.log("n2=" + req.query.n2) let n1 = parseFloat(req.query.n1) let n2 = parseFloat(req.query.n2) let result = n1 * n2 if (isNaN(result)) { result = "Valores inválidos." } else { result = `${req.query.n1} * ${req.query.n2} = ${result}` } res.send({ conta: result }) }) app.get('/divisao', (req, res) => { console.log("n1=" + req.query.n1) console.log("n2=" + req.query.n2) let n1 = parseFloat(req.query.n1) let n2 = parseFloat(req.query.n2) let result if (isNaN(n1) || isNaN(n2)) { result = "Valores inválidos." } else if (n2 === 0) { result = "Erro: Não é possível realizar divisão por zero." } else { result = `${req.query.n1} / ${req.query.n2} = ${n1 / n2}` } res.send({ conta: result }) }) app.listen(8080, () => { console.log('Servidor rodando na porta 8080') })
